Part 2: Visual Analysis with Python

We will create a script that takes a Base64 encoded image and sends it to our local model.

import requests
import base64

def image_to_base64(image_path):
    with open(image_path, "rb") as image_file:
        return base64.b64encode(image_file.read()).decode('utf-8')

def analyze_image(image_path, prompt):
    img_b64 = image_to_base64(image_path)
    url = "http://localhost:11434/api/generate"
    
    data = {
        "model": "moondream",
        "prompt": prompt,
        "images": [img_b64],
        "stream": False
    }
    
    response = requests.post(url, json=data)
    return response.json()['response']

# Try it out (ensure you have an image file handy)
print(analyze_image("circuit_diagram.jpg", "Identify the components in this diagram."))
